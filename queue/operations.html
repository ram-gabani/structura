Queue Operations & Modulo Algorithms

Learn how to implement both Linear and Circular Queue operations in standard C. Understand the boundary checks for Queue Overflow and Underflow, and master the modulo arithmetic that turns fixed arrays into seamless circular rings.

1. The Enqueue Operation (Insertion)

Adding an element to the rear of the circular queue

⚡ O(1) Time
Algorithmic Steps:
  1. Overflow Check: Check if (rear + 1) % MAX == front. If true, queue is full; abort.
  2. First Element Check: If front == -1 (empty queue), set front = 0.
  3. Advance Rear: Compute rear = (rear + 1) % MAX.
  4. Store Value: Set arr[rear] = value.
Circular Enqueue in C 
void enqueue(struct CircularQueue *q, int value) {
    /* Step 1: Check Overflow */
    if ((q->rear + 1) % MAX == q->front) {
        printf("Error: Circular Queue Overflow! Cannot insert %d.\n", value);
        return;
    }
    /* Step 2: Handle initial empty state */
    if (q->front == -1) {
        q->front = 0;
    }
    /* Step 3: Modulo advance and store */
    q->rear = (q->rear + 1) % MAX;
    q->arr[q->rear] = value;
    printf("Enqueued %d at Slot [%d].\n", value, q->rear);
}

2. The Dequeue Operation (Deletion)

Removing the earliest element from the front of the queue

⚡ O(1) Time
Algorithmic Steps:
  1. Underflow Check: Check if front == -1. If true, queue is empty; abort.
  2. Retrieve Value: Save val = arr[front].
  3. Final Element Check: If front == rear (only 1 element was left), reset queue to empty: front = -1; rear = -1;.
  4. Advance Front: Otherwise, advance via modulo: front = (front + 1) % MAX.
  5. Return val.
Circular Dequeue in C 
int dequeue(struct CircularQueue *q) {
    /* Step 1: Check Underflow */
    if (q->front == -1) {
        printf("Error: Circular Queue Underflow! Queue is empty.\n");
        return -1;
    }

    int val = q->arr[q->front];

    /* Step 2: If last remaining item is removed, reset to empty */
    if (q->front == q->rear) {
        q->front = -1;
        q->rear = -1;
    } else {
        /* Step 3: Modulo advance */
        q->front = (q->front + 1) % MAX;
    }

    return val;
}

Complete Circular Queue C Program

Fully compilable standard C99 program with struct CircularQueue

C99 circular_queue.c 
#include <stdio.h>
#include <stdlib.h>

#define MAX 5

struct CircularQueue {
    int arr[MAX];
    int front;
    int rear;
};

void initQueue(struct CircularQueue *q) {
    q->front = -1;
    q->rear = -1;
}

int isEmpty(struct CircularQueue *q) {
    return (q->front == -1);
}

int isFull(struct CircularQueue *q) {
    return ((q->rear + 1) % MAX == q->front);
}

void enqueue(struct CircularQueue *q, int val) {
    if (isFull(q)) {
        printf("Queue is FULL! Overflow.\n");
        return;
    }
    if (isEmpty(q)) {
        q->front = 0;
    }
    q->rear = (q->rear + 1) % MAX;
    q->arr[q->rear] = val;
    printf("Enqueued: %d\n", val);
}

int dequeue(struct CircularQueue *q) {
    if (isEmpty(q)) {
        printf("Queue is EMPTY! Underflow.\n");
        return -1;
    }
    int val = q->arr[q->front];
    if (q->front == q->rear) {
        /* Queue becomes empty */
        q->front = -1;
        q->rear = -1;
    } else {
        q->front = (q->front + 1) % MAX;
    }
    return val;
}

void display(struct CircularQueue *q) {
    if (isEmpty(q)) {
        printf("Queue is Empty.\n");
        return;
    }
    printf("Queue elements: ");
    int i = q->front;
    while (1) {
        printf("[%d] ", q->arr[i]);
        if (i == q->rear) break;
        i = (i + 1) % MAX;
    }
    printf("\n");
}

int main(void) {
    struct CircularQueue q;
    initQueue(&q);

    enqueue(&q, 10);
    enqueue(&q, 20);
    enqueue(&q, 30);
    display(&q);

    printf("Dequeued: %d\n", dequeue(&q));
    display(&q);

    enqueue(&q, 40);
    enqueue(&q, 50);
    enqueue(&q, 60); /* Wraps around to Slot 0! */
    display(&q);

    return 0;
}
